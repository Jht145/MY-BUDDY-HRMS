MY BUDDY HRMS — Next.js Frontend Starter
=========================================

This starter follows the project README: My Buddy HRMS with employee and HR Admin experiences, kiosk attendance UI, leave, payroll, profile, responsive navigation, and light/dark mode.

1. CREATE THE PROJECT
---------------------
Run these commands in your terminal:

npx create-next-app@latest my-buddy-hrms --ts --tailwind --eslint --app --src-dir --import-alias "@/*"
cd my-buddy-hrms
npm install lucide-react

Replace the generated files with the files below.

function AuthScreen({ mode, go, onAuth, theme, toggleTheme }: { mode: "signin" | "signup"; go: (page: Page) => void; onAuth: (role: Role, details?: Partial<AuthUser>) => void; theme: string; toggleTheme: () => void }) {
  const [error, setError] = useState("");
  const [loading, setLoading] = useState(false);
  const [showPassword, setShowPassword] = useState(false);
  const isSignUp = mode === "signup";

  function submit(formData: FormData) {
    setError("");
    const email = String(formData.get("email") || "").trim();
    const password = String(formData.get("password") || "");
    if (!/^\S+@\S+\.\S+$/.test(email)) return setError("Enter a valid work email address.");
    if (password.length < 8) return setError("Password must be at least 8 characters.");
    if (isSignUp && password !== String(formData.get("confirmPassword") || "")) return setError("Passwords do not match.");
    if (isSignUp && !String(formData.get("employeeId") || "").trim()) return setError("Employee ID is required.");
    setLoading(true);
    setTimeout(() => {
      const role = (formData.get("role") || "EMPLOYEE") as Role;
      onAuth(role, { name: String(formData.get("name") || "Priya Nair"), employeeId: String(formData.get("employeeId") || "EMP-1042"), email, company: String(formData.get("company") || "My Buddy Technologies") });
      setLoading(false);
    }, 650);
  }

  return <div className="min-h-screen bg-[var(--background)] p-4 sm:p-8"><header className="mx-auto flex max-w-5xl items-center justify-between"><button onClick={() => go("home")}><Brand /></button><button onClick={toggleTheme} className="rounded-xl border border-[var(--border)] p-2">{theme === "dark" ? <Sun size={18} /> : <Moon size={18} />}</button></header><div className="mx-auto mt-10 grid max-w-5xl overflow-hidden rounded-3xl border border-[var(--border)] bg-[var(--card)] shadow-2xl shadow-teal-950/10 lg:grid-cols-[0.9fr_1.1fr]"><div className="hidden bg-gradient-to-br from-[#0b1f3a] via-[#135c83] to-teal-500 p-10 text-white lg:block"><ShieldCheck size={42} /><h1 className="mt-10 text-4xl font-black">Work smarter.
